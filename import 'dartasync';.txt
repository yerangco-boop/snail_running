        return ThemeConfig(
          backgroundColor: const Color(0xFF0B0F19),
          primaryTextColor: const Color(0xFFE2E8F0),

class WorkoutRecord {
  final int? id;
  final String date; // ISO8601 String
  final int durationSeconds;
  final double distanceKm;
  final String averagePace;
  final int bpm;
  final String painAreas; // e.g. "무릎,종아리" or "없음"
  final int midfootMaintained; // 1 for Yes, 0 for No

  WorkoutRecord({
    this.id,
    required this.date,
    required this.durationSeconds,
    required this.distanceKm,
    required this.averagePace,
    required this.bpm,
    required this.painAreas,
    required this.midfootMaintained,
  });

  Map<String, dynamic> toMap() {
    return {
      'id': id,
      'date': date,
      'duration_seconds': durationSeconds,
      'distance_km': distanceKm,
      'average_pace': averagePace,
      'bpm': bpm,
      'pain_areas': painAreas,
      'midfoot_maintained': midfootMaintained,
    };
  }

  factory WorkoutRecord.fromMap(Map<String, dynamic> map) {
    return WorkoutRecord(
      id: map['id'],
      date: map['date'],
      durationSeconds: map['duration_seconds'],
      distanceKm: map['distance_km'],
      averagePace: map['average_pace'],
      bpm: map['bpm'],
      painAreas: map['pain_areas'] ?? '없음',
      midfootMaintained: map['midfoot_maintained'] ?? 1,
    );
  }
}

class DatabaseHelper {
  static final DatabaseHelper instance = DatabaseHelper._init();
  static Database? _database;

  DatabaseHelper._init();

  Future<Database> get database async {
    if (_database != null) return _database!;
    _database = await _initDB('snail_running_db.db');
    return _database!;
  }

  Future<Database> _initDB(String filePath) async {
    final dbPath = await getDatabasesPath();
    final path = p.join(dbPath, filePath);
    return await openDatabase(
      path,
      version: 2,
      onCreate: _createDB,
      onUpgrade: _upgradeDB,
    );
  }

  Future<void> _createDB(Database db, int version) async {
    await db.execute('''
      CREATE TABLE workout_records (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        date TEXT NOT NULL,
        duration_seconds INTEGER NOT NULL,
        distance_km REAL NOT NULL,
        average_pace TEXT NOT NULL,
        bpm INTEGER NOT NULL,
        pain_areas TEXT NOT NULL,
        midfoot_maintained INTEGER NOT NULL DEFAULT 1
      )
    ''');
  }

  Future<void> _upgradeDB(Database db, int oldVersion, int newVersion) async {
    if (oldVersion < 2) {
      // Add midfoot_maintained column if upgrading from v1
      try {
        await db.execute('ALTER TABLE workout_records ADD COLUMN midfoot_maintained INTEGER NOT NULL DEFAULT 1');
      } catch (e) {
        debugPrint("Column might already exist: $e");
      }
    }
  }

  Future<int> insertRecord(WorkoutRecord record) async {
    final db = await instance.database;
    return await db.insert('workout_records', record.toMap());
  }

  Future<List<WorkoutRecord>> fetchAllRecords() async {
    final db = await instance.database;
    final result = await db.query('workout_records', orderBy: 'date DESC');
    return result.map((json) => WorkoutRecord.fromMap(json)).toList();
  }

  Future<int> clearAllRecords() async {
    final db = await instance.database;
    return await db.delete('workout_records');
  }
}

class NavigationScreen extends StatefulWidget {
  const NavigationScreen({super.key});

  @override
  State<NavigationScreen> createState() => _NavigationScreenState();
}

class _NavigationScreenState extends State<NavigationScreen> {
  int _currentIndex = 0;
  AppTheme _theme = AppTheme.oledBlack;

  // Track key references to refresh pages
  final GlobalKey<_HistoryStatsTabState> _historyKey = GlobalKey();

  void _onThemeChanged(AppTheme newTheme) {
    setState(() {
      _theme = newTheme;
    });
  }

  @override
  Widget build(BuildContext context) {
    final config = ThemeConfig.from(_theme);
    
    final List<Widget> tabs = [
      MainRunningTab(
        config: config,
        theme: _theme,
        onThemeChanged: _onThemeChanged,
        onRecordSaved: () {
          // Refresh statistics
          _historyKey.currentState?.loadRecords();
        },
      ),
      HistoryStatsTab(
        key: _historyKey,
        config: config,
      ),
    ];

    return Scaffold(
      backgroundColor: config.backgroundColor,
      body: tabs[_currentIndex],
      bottomNavigationBar: BottomNavigationBar(
        currentIndex: _currentIndex,
        onTap: (index) {
          setState(() {
            _currentIndex = index;
          });
        },
        backgroundColor: config.cardBackgroundColor,
        selectedItemColor: config.accentColor,
        unselectedItemColor: config.secondaryTextColor,
        showUnselectedLabels: true,
        type: BottomNavigationBarType.fixed,
        items: const [
          BottomNavigationBarItem(
            icon: Icon(Icons.directions_run_rounded),
            label: '러닝 가이드',
          ),
          BottomNavigationBarItem(
            icon: Icon(Icons.bar_chart_rounded),
            label: '기록 & 통계',
          ),
        ],
      ),
    );
  }
}

class MainRunningTab extends StatefulWidget {
  final ThemeConfig config;
  final AppTheme theme;
  final ValueChanged<AppTheme> onThemeChanged;
  final VoidCallback onRecordSaved;

  const MainRunningTab({
    super.key,
    required this.config,
    required this.theme,
    required this.onThemeChanged,
    required this.onRecordSaved,
  });

  @override
  State<MainRunningTab> createState() => _MainRunningTabState();
}

class _MainRunningTabState extends State<MainRunningTab> with SingleTickerProviderStateMixin {
  // Timer & workout metrics
  bool _isRunning = false;
  bool _isPaused = false;
  int _seconds = 0;
  double _distanceKm = 0.0;
  Timer? _workoutTimer;

  // Metronome: 150 BPM (400ms interval)
  bool _isMetronomeOn = false;
  Timer? _metronomeTimer;
  late AnimationController _pulseController;
  late Animation<double> _pulseAnimation;

  // Dynamic layout toggle
  FocusMetric _focusMetric = FocusMetric.time;

  @override
  void initState() {
    super.initState();
    _pulseController = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 150),
    );
    _pulseAnimation = Tween<double>(begin: 1.0, end: 1.25).animate(
      CurvedAnimation(parent: _pulseController, curve: Curves.easeOut),
    );
  }

  @override
  void dispose() {
    _workoutTimer?.cancel();
    _metronomeTimer?.cancel();
    _pulseController.dispose();
    super.dispose();
  }

  void _startWorkout() {
    setState(() {
      _isRunning = true;
      _isPaused = false;
    });
    _startMetronome();

    _workoutTimer = Timer.periodic(const Duration(seconds: 1), (timer) {
      setState(() {
        _seconds++;
        // Simulate a slow jog speed: ~8:00 min/km pace (~2.08 meters per second)
        _distanceKm += 0.00208;
      });
    });
  }

  void _pauseWorkout() {
    _workoutTimer?.cancel();
    _stopMetronome();
    setState(() {
      _isPaused = true;
    });
  }

  void _resumeWorkout() {
    setState(() {
      _isPaused = false;
    });
    _startWorkout();
  }

  void _startMetronome() {
    _metronomeTimer?.cancel();
    setState(() {
      _isMetronomeOn = true;
    });

    _metronomeTimer = Timer.periodic(const Duration(milliseconds: 400), (timer) {
      if (_isMetronomeOn) {
        SystemSound.play(SystemSoundType.click);
        _pulseController.forward().then((value) => _pulseController.reverse());
      }
    });
  }

  void _stopMetronome() {
    _metronomeTimer?.cancel();
    setState(() {
      _isMetronomeOn = false;
    });
  }

  String _formatTime(int totalSeconds) {
    int hrs = totalSeconds ~/ 3600;
    int mins = (totalSeconds % 3600) ~/ 60;
    int secs = totalSeconds % 60;
    String hrsStr = hrs > 0 ? '${hrs.toString().padLeft(2, '0')}:' : '';
    return '$hrsStr${mins.toString().padLeft(2, '0')}:${secs.toString().padLeft(2, '0')}';
  }

  String _getPace() {
    if (_distanceKm == 0) return "--'--\"";
    double minutes = (_seconds / 60.0) / _distanceKm;
    int paceMin = minutes.floor();
    int paceSec = ((minutes - paceMin) * 60).round();
    return "$paceMin'${paceSec.toString().padLeft(2, '0')}\"";
  }

  List<WorkoutRecord> _records = [];
  bool _isLoading = true;
  String _statsFilter = 'weekly'; // 'daily', 'weekly', 'monthly'

  @override
  void initState() {
    super.initState();
    loadRecords();
  }

  Future<void> loadRecords() async {
    setState(() => _isLoading = true);
    final data = await DatabaseHelper.instance.fetchAllRecords();
    setState(() {
      _records = data;
      _isLoading = false;
    });
  }

  // Calculate statistics grouping
  Map<String, double> _calculateTimeStats() {
    final now = DateTime.now();
    Map<String, double> grouped = {};

    for (var record in _records) {
      final date = DateTime.tryParse(record.date) ?? now;
      String key = '';
      if (_statsFilter == 'daily') {
        // Last 7 days individually
        key = '${date.month}/${date.day}';
        final diff = now.difference(date).inDays;
        if (diff > 7) continue;
      } else if (_statsFilter == 'weekly') {
        // Weeks of this year
        final weekNum = (date.difference(DateTime(date.year, 1, 1)).inDays / 7).ceil();
        key = '${date.year}년 $weekNum주차';
        final diff = now.difference(date).inDays;
        if (diff > 35) continue; // limit to last 5 weeks
      } else {
        // Months
        key = '${date.year}/${date.month}';
        final diff = now.difference(date).inDays;
        if (diff > 180) continue; // limit to 6 months
      }

      grouped[key] = (grouped[key] ?? 0.0) + (record.durationSeconds / 60.0); // minutes
    }
    return grouped;
  }

  // Pain analysis stats
  Map<String, int> _calculatePainStats() {
    Map<String, int> painCounts = {};
    for (var record in _records) {
      if (record.painAreas != '없음') {
        final areas = record.painAreas.split(',');
        for (var area in areas) {
          if (area.trim().isNotEmpty) {
            painCounts[area] = (painCounts[area] ?? 0) + 1;
          }
        }
      }
    }
    return painCounts;
  }

  // Mock Google Sheets Sync Skeleton
  Future<void> _syncToGoogleSheets() async {
    // 1. Convert local database records to JSON
    List<Map<String, dynamic>> payload = _records.map((r) => r.toMap()).toList();
    
    // 2. Mock Webhook/Apps Script URL
    // (In production, replace with real Apps Script Deployment URL)
    const String appsScriptUrl = 'https://script.google.com/macros/s/MOCK_DEPLOYMENT_ID/exec';

    if (payload.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('동기화할 러닝 기록이 없습니다.')),
      );
      return;
    }

    try {
      // Show loader
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('구글 시트와 연동/전송을 시작합니다...')),
      );

      // Create JSON payload
      final jsonPayload = jsonEncode({
        'sync_date': DateTime.now().toIso8601String(),
        'records': payload,
      });

      debugPrint('--- GOOGLE SHEETS SYNC PAYLOAD ---');
      debugPrint(jsonPayload);
      debugPrint('----------------------------------');

      // Dart Native HttpClient call simulation (skeleton)
      // Since it's a mock endpoint, we handle a short timeout or mock response.
      // Set to true to mock behavior to prevent HTTP connection crashes in test env
      const bool isMock = true; 
      if (!isMock) {
        final client = HttpClient();
        final request = await client.postUrl(Uri.parse(appsScriptUrl));
        request.headers.set('content-type', 'application/json');
        request.add(utf8.encode(jsonPayload));
        final response = await request.close();
        debugPrint('Response Status: ${response.statusCode}');
      }

      await Future.delayed(const Duration(seconds: 1)); // simulated network latency

      if (mounted) {
        showDialog(
          context: context,
          builder: (context) => AlertDialog(
            backgroundColor: widget.config.cardBackgroundColor,
            title: const Text('✅ 구글 시트 동기화 완료 (뼈대)'),
            content: const Text(

              '실사용 시 Apps Script URL을 연동하시면 구글 시트에 자동으로 저장됩니다.',
